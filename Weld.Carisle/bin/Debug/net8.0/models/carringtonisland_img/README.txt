  c_carisle_mid_seabed_00.fbx      <DffFile>c_carisle_mid_seabed_00</DffFile>  + ".fbx"  (AssimpFbxModelFormat)
  c_carisle_mid_seabed_01.fbx
  c_carisle_mid_seabed_02.fbx
  generictxd/texture.png           <TextureDictionary>generic</TextureDictionary> + "txd/texture.png"  (PngTextureDictionaryFormat)
